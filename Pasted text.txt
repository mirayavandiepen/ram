"use client";

import { useRef, useState } from "react";
import {
  animate,
  motion,
  useIsomorphicLayoutEffect,
  useMotionValue,
  useReducedMotion,
} from "motion/react";

import { EASE_IN_OUT, EASE_OUT, seconds, TIMELINE } from "@/lib/motion";

/**
 * Tracking, in em, as an offset from whatever the heading already sets rather
 * than as a value of its own. Loose, then well past tight, then back to exactly
 * where it started.
 */
const TRACKING = [0, 0.026, -0.07, 0];
const TRACKING_TIMES = [0, 0.3, 0.63, 1];

/** Sub-pixel churn is not a resize, and re-rendering on it would never settle. */
const RESIZE_EPSILON = 0.5;

/**
 * One canvas for the life of the page. `measure` runs on mount, again on the
 * beat and again on every resize, and each call was building a fresh element and
 * a fresh 2D context to read two numbers that only depend on the font.
 * `undefined` means not yet attempted, `null` means attempted and unavailable.
 */
let metricsContext: CanvasRenderingContext2D | null | undefined;

/**
 * How much room the line box gave up, in px per side.
 *
 * Display leading runs tighter than the glyphs actually need, so a box drawn
 * flush to the line box cuts the tail off the g. CSS centres the font's content
 * area inside the line box, which makes the shortfall symmetric, so half the
 * difference is exactly the amount to bleed by on each side. Asking the font
 * rather than hard-coding a number means the heading's leading can be retuned
 * without cropping the word.
 *
 * Returns 0 where the metrics are missing, which just draws the flush box rather
 * than guessing at one.
 */
function contentOverhang(word: HTMLElement, lineBox: number) {
  if (metricsContext === undefined) {
    metricsContext = document.createElement("canvas").getContext("2d");
  }
  const context = metricsContext;
  if (!context) return 0;

  const style = getComputedStyle(word);
  context.font = `${style.fontWeight} ${style.fontSize} ${style.fontFamily}`;
  // These two are properties of the font, not of the string, so what gets
  // passed here does not matter as long as it is not empty.
  const metrics = context.measureText("x");
  const content =
    (metrics.fontBoundingBoxAscent ?? 0) + (metrics.fontBoundingBoxDescent ?? 0);

  return Math.max(0, (content - lineBox) / 2);
}

/**
 * How far the outline sits outside the glyphs. Wider on the right because the
 * word ends in a curve and opens on a stem, so a symmetric inset would land the
 * stroke on the last letter. Kept here rather than in a class so the badge can
 * be centred on the same numbers the outline is drawn with.
 */
const INSET = { left: 4, right: 8 };

/** The four handles, each pulled back by half its size so it straddles a corner. */
const HANDLES = [
  "top-[-3px] left-[-3px]",
  "top-[-3px] right-[-3px]",
  "bottom-[-3px] left-[-3px]",
  "bottom-[-3px] right-[-3px]",
];

type Size = { width: number; height: number; bleed: number };

const formatSize = (width: number, height: number) =>
  `${Math.round(width)} × ${Math.round(height)}`;

type SelectionBoxProps = {
  /**
   * One word. A string rather than nodes: the badge predicts its own width from
   * the character count, and split text sets wider than the same run in a
   * single text node.
   */
  children: string;
  /**
   * Milliseconds from mount before the box is drawn. Passed in rather than read
   * from the timeline, because it has to wait out the caret on the row below and
   * only the caller knows how long that copy takes to write and retire.
   */
  delay: number;
};

/**
 * Draws a design-tool selection box around `children`: a thin outline, four
 * corner handles, and a live `width × height` badge.
 *
 * The box is transient. It arrives, takes its measurement, holds it long enough
 * to be read, and then leaves the page the way the caret does, so the copy is
 * the only thing still standing at the end.
 *
 * The word is never pinned to a fixed width. Its own box grows and shrinks with
 * the tracking, which is exactly what lets the outline and the badge track it.
 * What is held constant is the width of the *line*: a compensating margin gives
 * back whatever the tracking took, so the heading's total advance never changes.
 * Without it the block, which is shrink-to-fit and centred as a whole, would
 * re-centre on every frame and drag the line below sideways for a change it has
 * no part in.
 */
export function SelectionBox({ children, delay }: SelectionBoxProps) {
  const wordRef = useRef<HTMLSpanElement>(null);
  const badgeRef = useRef<HTMLSpanElement>(null);
  const [size, setSize] = useState<Size | null>(null);
  const [dismissed, setDismissed] = useState(false);
  const opacity = useMotionValue(0);
  const shouldReduceMotion = useReducedMotion();

  useIsomorphicLayoutEffect(() => {
    const word = wordRef.current;
    if (!word) return;

    // Nothing to draw. Now that the box leaves on its own, a reduced-motion
    // page that kept it would be the only one ending with a box on it, which is
    // a different page rather than a calmer one. The caret is already handled
    // this way: what is only ever transient simply never arrives.
    if (shouldReduceMotion) return;

    const measure = () => {
      const rect = word.getBoundingClientRect();
      const bleed = contentOverhang(word, rect.height);
      const next = {
        width: rect.width,
        height: rect.height + bleed * 2,
        bleed,
      };
      setSize((previous) =>
        previous &&
        Math.abs(previous.width - next.width) < RESIZE_EPSILON &&
        Math.abs(previous.height - next.height) < RESIZE_EPSILON
          ? previous
          : next,
      );
      return next;
    };

    // Attached only once the word is at rest. During the tracking pass this
    // would fire on every frame, and every frame would be a re-render. It also
    // fires once on connect, which re-syncs the badge against anything that
    // moved while the sequence was running, a late webfont swap being the one
    // that actually happens.
    let observer: ResizeObserver | undefined;
    const observeAtRest = () => {
      observer = new ResizeObserver(measure);
      observer.observe(word);
    };

    measure();

    const running: { stop: () => void }[] = [];
    const timers: number[] = [];

    // Holds the final measurement, then takes the box off the page and out of
    // the DOM. Fading it to nothing and leaving it mounted would keep a stale
    // observer and an invisible layer over the word for good.
    const leave = () => {
      running.push(
        animate(opacity, 0, {
          duration: seconds(TIMELINE.selectionExitDuration),
          ease: EASE_OUT,
          onComplete: () => {
            observer?.disconnect();
            setDismissed(true);
          },
        }),
      );
    };

    const start = () => {
      // Read on the beat rather than at mount. The webfont swaps in between the
      // two, and a baseline taken against the fallback would have the badge
      // reporting a number the word never had for the whole pass.
      const rest = measure();
      const style = getComputedStyle(word);
      // Reads "normal" when nothing has set it, which parses to NaN.
      const restSpacing = Number.parseFloat(style.letterSpacing) || 0;
      const fontSize = Number.parseFloat(style.fontSize);

      // Freeze the label's box at its resting width. The reported number loses a
      // digit at the tight extreme, and a chip that resizes around a centred
      // origin shifts by half a digit when it does. Pinning the width lets the
      // text recentre inside a box that never moves. One layout read, taken here
      // on the beat rather than anywhere near the per-frame path.
      const badge = badgeRef.current;
      if (badge) {
        badge.textContent = formatSize(rest.width, rest.height);
        badge.style.width = `${badge.getBoundingClientRect().width}px`;
      }

      running.push(
        animate(opacity, 1, {
          duration: seconds(TIMELINE.selectionFadeDuration),
          ease: EASE_OUT,
        }),
        animate(0, TRACKING, {
          delay: seconds(
            TIMELINE.selectionFadeDuration - TIMELINE.selectionTrackingLead,
          ),
          duration: seconds(TIMELINE.selectionTrackingDuration),
          times: TRACKING_TIMES,
          ease: [EASE_IN_OUT, EASE_IN_OUT, EASE_OUT],
          onUpdate: (offset) => {
            const offsetPx = offset * fontSize;
            // letter-spacing lands after every character including the last, so
            // the width the word gains is the offset times the whole count.
            const delta = offsetPx * children.length;
            word.style.letterSpacing = `${restSpacing + offsetPx}px`;
            // Hand the delta straight back, so the line's total advance is
            // unchanged and nothing below this row moves. At the loose extreme
            // the glyphs spill past the heading's content edge, which is the
            // other reason nothing from here up may clip.
            word.style.marginRight = `${-delta}px`;
            // Predicted, never measured, so no layout read reaches the
            // per-frame path.
            if (badgeRef.current) {
              badgeRef.current.textContent = formatSize(
                rest.width + delta,
                rest.height,
              );
            }
          },
          onComplete: () => {
            // Back to the heading's own tracking, and the compensation goes
            // with it. A literal em value would silently drop that tracking,
            // and a frozen px value would leave the word mistracked at every
            // width but this one.
            word.style.letterSpacing = "";
            word.style.marginRight = "";
            observeAtRest();
            timers.push(window.setTimeout(leave, TIMELINE.selectionRest));
          },
        }),
      );
    };

    timers.push(window.setTimeout(start, delay));

    return () => {
      for (const timer of timers) window.clearTimeout(timer);
      for (const animation of running) animation.stop();
      observer?.disconnect();
      word.style.letterSpacing = "";
      word.style.marginRight = "";
      if (badgeRef.current) badgeRef.current.style.width = "";
    };
  }, [children, delay, opacity, shouldReduceMotion]);

  return (
    <span ref={wordRef} className="relative inline-block">
      {children}
      {/* Nothing until the word has been measured, so there is never a frame
          with the box drawn somewhere it does not belong, and nothing again once
          it has left. */}
      {size && !dismissed ? (
        <motion.span
          aria-hidden="true"
          style={{
            opacity,
            top: -size.bleed,
            bottom: -size.bleed,
            left: -INSET.left,
            right: -INSET.right,
          }}
          // Decoration throughout: never in the accessibility tree, never takes
          // a click, and never comes along when the heading is selected.
          //
          // `tracking-normal` is load-bearing. The tracking being animated is an
          // inline style on the word, and letter-spacing inherits, so without a
          // declaration of its own the badge would be handed the word's tracking
          // in px: roughly -4px on 11px text at the tight extreme, which crushes
          // the label into itself. Declaring it here stops the animation at the
          // chrome boundary.
          className="pointer-events-none absolute tracking-normal select-none"
        >
          {/* The outline is a child rather than a border on the layer itself,
              so the handles and the badge measure from the stroke instead of
              from a padding box already inset by its width. A hairline: at 1px
              the box reports the measurement without competing with the word it
              is measuring. Held at 60% so the guide stays subordinate to the
              word: at full strength it is exactly as bright as the heading and
              starts reading as part of it. */}
          <span className="absolute inset-0 border border-foreground/60" />
          {HANDLES.map((position) => (
            // Solid, and the only part of the box at full strength, because the
            // handles are the thing you would grab. Filled rather than outlined
            // now that they share the foreground: white in dark mode, near-black
            // in light, so unlike a fixed white fill they never disappear into
            // the page.
            <span
              key={position}
              className={`absolute size-[6px] bg-foreground ${position}`}
            />
          ))}
          <span
            ref={badgeRef}
            // Set in the page's own sans, not mono. Figma's own badge is sans,
            // and a monospace stack would hand this one label to whatever the
            // OS ships. Tabular figures do the work mono was there for.
            // (Naming the utility here would be enough for Tailwind to scan it
            // out of this comment and emit the unused class.)
            //
            // White on --chrome-label is 7.77:1, so this clears AA on its own
            // and needs no exemption. The blue it replaced sat at 3.64:1 and did.
            //
            // Pinned to the outline's resting centre rather than to `left-1/2`.
            // A percentage would be resolved against the layer, which is as wide
            // as the word currently is, so the label would slide back and forth
            // for the whole pass. `size` is the measurement taken at rest and is
            // not touched again, so this holds still while the box breathes.
            style={{ left: (size.width + INSET.left + INSET.right) / 2 }}
            className="absolute bottom-full mb-[6px] -translate-x-1/2 rounded-[3px] bg-chrome-label px-[5px] py-[1px] text-center text-[11px] leading-[14px] font-medium whitespace-nowrap text-white tabular-nums"
          >
            {formatSize(size.width, size.height)}
          </span>
        </motion.span>
      ) : null}
    </span>
  );
}
